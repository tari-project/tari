# Valgrind Suppressions for Tari Wallet FFI
#
# This file contains suppressions for known false positives in memory leak detection
# when running Tari wallet FFI callback tests under valgrind.

# Rust standard library allocations
{
   rust_std_alloc
   Memcheck:Leak
   match-leak-kinds: possible
   fun:malloc
   ...
   fun:rust_oom
}

   rust_std_vec_alloc
   Memcheck:Leak

   fun:malloc
   ...
   obj:*/libstd-*
